"""Unicode normalization utilities for the account service.

Provides functions for normalizing usernames to consistent forms
for storage, lookup, and display.
"""
import unicodedata
from account_service.config import NORMALIZATION_FORM


def normalize_for_storage(username):
    """Normalizes usernames for consistent storage and lookup.

    Reads the configured normalization form from config.py and applies
    it to the input string. This is the primary function that should be
    called when usernames enter the system.

    Args:
        username: The raw username string to normalize.

    Returns:
        The normalized username string, or the input unchanged if empty.
    """
    if not username:
        return username
    return unicodedata.normalize(NORMALIZATION_FORM, username)


def normalize_for_display(username):
    """Normalizes usernames for display purposes.

    Uses NFC form which is the canonical storage form for this application.
    Display normalization ensures that visually identical characters are
    rendered consistently regardless of internal representation.

    Args:
        username: The username string to normalize for display.

    Returns:
        The NFC-normalized username string, or the input unchanged if empty.
    """
    if not username:
        return username
    return unicodedata.normalize('NFC', username)


def is_normalized(username, form=None):
    """Check if a username is already in the specified normalization form.

    Useful for validation and debugging to determine whether a string
    needs further normalization.

    Args:
        username: The string to check.
        form: The normalization form to check against. Defaults to the
              configured NORMALIZATION_FORM.

    Returns:
        True if the username is already in the specified form.
    """
    if not username:
        return True
    if form is None:
        form = NORMALIZATION_FORM
    return unicodedata.is_normalized(form, username)


def get_normalization_form():
    """Return the currently configured normalization form.

    Returns:
        The string identifier of the active normalization form
        (e.g., 'NFC', 'NFD', 'NFKC', 'NFKD').
    """
    return NORMALIZATION_FORM


def compare_normalized(username_a, username_b, form=None):
    """Compare two usernames after normalization.

    Two usernames that look identical on screen may have different
    internal representations. This function normalizes both and
    checks for equality.

    Args:
        username_a: First username to compare.
        username_b: Second username to compare.
        form: Normalization form to use. Defaults to configured form.

    Returns:
        True if both usernames are equal after normalization.
    """
    if form is None:
        form = NORMALIZATION_FORM
    a = unicodedata.normalize(form, username_a) if username_a else ''
    b = unicodedata.normalize(form, username_b) if username_b else ''
    return a == b


def detect_form(username):
    """Detect which normalization form a username is likely in.

    Checks NFC and NFD forms and returns a list of forms that
    the string matches.

    Args:
        username: The string to analyze.

    Returns:
        A list of normalization form strings that the username matches.
    """
    if not username:
        return ['NFC', 'NFD', 'NFKC', 'NFKD']
    forms = []
    for f in ['NFC', 'NFD', 'NFKC', 'NFKD']:
        if unicodedata.is_normalized(f, username):
            forms.append(f)
    return forms

"""Username validation and cleaning utilities.

Provides functions for sanitizing, validating, and cleaning usernames
before they are stored in the database. The cleaning pipeline consists
of several stages:

1. sanitize_username: Strip whitespace and lowercase
2. _canonicalize: Apply Unicode normalization (via _internal module)
3. clean_username: Combines sanitize + canonicalize

The validation functions check format requirements such as length
and character restrictions.
"""
import re
from account_service._internal import _canonicalize
from account_service.config import MAX_USERNAME_LENGTH, MIN_USERNAME_LENGTH


def sanitize_username(username):
    """Strip whitespace and lowercase the username.

    This performs basic sanitization: removing leading/trailing
    whitespace and converting to lowercase. It does not perform
    Unicode normalization — that is handled by _canonicalize().

    Args:
        username: The raw username string.

    Returns:
        The sanitized username, or empty string if input is falsy.
    """
    if not username:
        return ''
    return username.strip().lower()


def clean_username(username):
    """Fully clean and normalize a username for storage.

    Applies the complete cleaning pipeline:
    1. sanitize_username (strip + lowercase)
    2. _canonicalize (Unicode normalization)

    This should be used when a username first enters the system
    through user-facing entry points such as registration.

    Args:
        username: The raw username string.

    Returns:
        The cleaned and normalized username.
    """
    sanitized = sanitize_username(username)
    return _canonicalize(sanitized)


def validate_username(username):
    """Validate a username meets format requirements.

    Checks:
    - Not empty or whitespace-only
    - Does not exceed maximum length
    - Does not contain control characters
    - Does not contain prohibited patterns

    Args:
        username: The username string to validate (should already be
                 sanitized/lowercased).

    Returns:
        A tuple of (is_valid: bool, error_message: str or None).
    """
    if not username or not username.strip():
        return False, "Username cannot be empty or whitespace"

    if len(username) < MIN_USERNAME_LENGTH:
        return False, f"Username must be at least {MIN_USERNAME_LENGTH} character(s)"

    if len(username) > MAX_USERNAME_LENGTH:
        return False, f"Username exceeds maximum length of {MAX_USERNAME_LENGTH}"

    # Reject control characters
    if re.search(r'[\x00-\x1f\x7f]', username):
        return False, "Username contains control characters"

    # Reject usernames that are only punctuation or symbols
    if re.match(r'^[\W_]+$', username) and not any(c.isalpha() for c in username):
        return False, "Username must contain at least one letter or digit"

    return True, None


def validate_username_strict(username):
    """Validate a username with additional strictness checks.

    In addition to the basic validation, also checks for:
    - No consecutive special characters
    - No leading or trailing special characters (after whitespace removal)

    Args:
        username: The username string to validate.

    Returns:
        A tuple of (is_valid: bool, error_message: str or None).
    """
    is_valid, error = validate_username(username)
    if not is_valid:
        return is_valid, error

    # Check for consecutive non-alphanumeric characters
    if re.search(r'[^a-zA-Z0-9]{2,}', username):
        return False, "Username cannot contain consecutive special characters"

    return True, None


def is_ascii_only(username):
    """Check if a username contains only ASCII characters.

    Args:
        username: The username string to check.

    Returns:
        True if all characters are ASCII (code points 0-127).
    """
    if not username:
        return True
    try:
        username.encode('ascii')
        return True
    except UnicodeEncodeError:
        return False

"""Internal normalization utilities.

This module provides low-level Unicode normalization functions used
by the validation module. It is not intended to be imported directly
by application code; use validation.clean_username() or
normalize.normalize_for_storage() instead.
"""
import unicodedata


def _canonicalize(username):
    """Apply canonical Unicode normalization to a username.

    This ensures that visually identical characters are represented
    consistently in their decomposed form for storage. The decomposition
    form is used because it preserves the logical structure of characters,
    making it easier to compare and transform usernames.

    Args:
        username: The username string to canonicalize.

    Returns:
        The canonicalized username string, or the input unchanged if empty.
    """
    if not username:
        return username
    return unicodedata.normalize('NFD', username)


def _decompose_char(char):
    """Decompose a single character into its component parts.

    Useful for debugging and inspecting how characters are represented
    internally.

    Args:
        char: A single character string.

    Returns:
        The decomposed form of the character.
    """
    return unicodedata.normalize('NFD', char)


def _compose_char(char):
    """Compose a single character from its decomposed parts.

    Args:
        char: A character string, possibly in decomposed form.

    Returns:
        The composed (NFC) form of the character.
    """
    return unicodedata.normalize('NFC', char)

"""WSGI middleware for the account service.

Provides input sanitization middleware that normalizes URL paths
to ensure consistent routing for Unicode characters.
"""
import unicodedata


class InputSanitizationMiddleware:
    """Middleware that normalizes URL paths for consistent routing.

    Ensures that Unicode characters in URL paths are in NFC form
    for consistent matching and routing. This handles the case where
    different clients may encode the same visual path differently.

    Note: This middleware only normalizes PATH_INFO in the WSGI environ.
    Request body data (JSON, form-encoded) is not affected and must be
    normalized separately at the route handler level.
    """

    def __init__(self, app):
        """Initialize the middleware with the WSGI application.

        Args:
            app: The WSGI application to wrap.
        """
        self.app = app

    def __call__(self, environ, start_response):
        """Process the request through the middleware.

        Normalizes PATH_INFO to NFC form before passing the request
        to the wrapped application.

        Args:
            environ: WSGI environment dictionary.
            start_response: WSGI start_response callable.

        Returns:
            The response from the wrapped application.
        """
        path_info = environ.get('PATH_INFO', '')
        if path_info:
            normalized = unicodedata.normalize('NFC', path_info)
            environ['PATH_INFO'] = normalized

        # Also normalize SCRIPT_NAME if present
        script_name = environ.get('SCRIPT_NAME', '')
        if script_name:
            environ['SCRIPT_NAME'] = unicodedata.normalize('NFC', script_name)

        return self.app(environ, start_response)


class RequestLoggingMiddleware:
    """Middleware that logs basic request information.

    Logs the HTTP method, path, and response status for each request.
    Useful for debugging and monitoring.
    """

    def __init__(self, app, logger=None):
        """Initialize with the WSGI application and optional logger.

        Args:
            app: The WSGI application to wrap.
            logger: Optional logging.Logger instance. If None, a
                   basic logger is created.
        """
        self.app = app
        self.logger = logger

    def __call__(self, environ, start_response):
        """Process and log the request."""
        method = environ.get('REQUEST_METHOD', 'UNKNOWN')
        path = environ.get('PATH_INFO', '/')

        def logging_start_response(status, headers, exc_info=None):
            if self.logger:
                self.logger.info(f"{method} {path} -> {status}")
            return start_response(status, headers, exc_info)

        return self.app(environ, logging_start_response)

"""Database access layer for the account service.

Provides functions for CRUD operations on accounts and balance
history, as well as configuration management via the app_config table.

All database connections use WAL mode for better concurrent read
performance and PRAGMA foreign_keys=ON for referential integrity.
"""
import sqlite3
from account_service.config import DB_PATH


def get_connection():
    """Get a database connection with row factory and recommended pragmas.

    Returns:
        A sqlite3.Connection object configured with:
        - Row factory set to sqlite3.Row for dict-like access
        - WAL journal mode for better concurrency
        - Foreign keys enabled
    """
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")
    conn.execute("PRAGMA busy_timeout=5000")
    return conn


def get_account_by_username(username):
    """Look up an account by exact username match.

    Args:
        username: The username to search for. Must match exactly,
                 including Unicode normalization form.

    Returns:
        A dict with account fields, or None if not found.
    """
    conn = get_connection()
    try:
        row = conn.execute(
            "SELECT * FROM accounts WHERE username = ?",
            (username,)
        ).fetchone()
        return dict(row) if row else None
    finally:
        conn.close()


def get_account_by_username_ci(username):
    """Look up an account by case-insensitive username match.

    Uses LOWER() for case-insensitive comparison while still
    requiring exact Unicode character match. This handles the
    case where usernames differ only in case but not in their
    Unicode representation.

    Args:
        username: The username to search for (case-insensitive).

    Returns:
        A dict with account fields, or None if not found.

    conn = get_connection()
    try:
        row = conn.execute(

        conn.close()

"""Flask application factory for the account service.

The application is created via the create_app() factory function,
which loads configuration in the following order:

1. Default values from config.py
2. Runtime overrides from the app_config database table
3. Optional config_overrides dict parameter

The database config override mechanism allows changing settings
without redeploying the application code.
"""
import os
import sys
import logging
from flask import Flask
from account_service.config import DB_PATH, SECRET_KEY, NORMALIZATION_FORM as CONFIG_NORMALIZATION_FORM
from account_service.middleware import InputSanitizationMiddleware
from account_service.error_handlers import register_error_handlers
from account_service.db import get_config_value


logger = logging.getLogger(__name__)


def create_app(config_overrides=None):
    """Create and configure the Flask application.

    Configuration is loaded in this order:
    1. Default config from config.py
    2. Runtime config from app_config database table (overrides config.py)
    3. Optional config_overrides dict (overrides everything)

    The database configuration allows dynamic changes without redeploying.
    When the app starts, it reads the app_config table and applies those
    values on top of the defaults.

    Args:
        config_overrides: Optional dict of configuration overrides.

    Returns:
        A configured Flask application instance.
    """
    app = Flask(__name__)

    # Step 1: Apply defaults from config.py
    app.config['SECRET_KEY'] = SECRET_KEY
    app.config['DB_PATH'] = DB_PATH
    app.config['NORMALIZATION_FORM'] = CONFIG_NORMALIZATION_FORM
    app.config['MAX_USERNAME_LENGTH'] = int(os.environ.get('MAX_USERNAME_LENGTH', '64'))
    app.config['MIN_BALANCE'] = float(os.environ.get('MIN_BALANCE', '-1000.0'))
    app.config['MAX_BALANCE'] = float(os.environ.get('MAX_BALANCE', '999999999.0'))

    # Step 2: Load runtime configuration from the database
    # These values take precedence over config.py defaults
    _load_db_config(app)

    # Step 3: Apply any programmatic overrides
    if config_overrides:
        app.config.update(config_overrides)

    # Sync the module-level config so normalize_for_storage picks it up
    import account_service.config as cfg
    cfg.NORMALIZATION_FORM = app.config['NORMALIZATION_FORM']

    # Set up logging
    if app.config.get('DEBUG'):
        logging.basicConfig(level=logging.DEBUG)
    else:
        logging.basicConfig(level=logging.INFO)

    logger.info(f"Account service starting with NORMALIZATION_FORM={app.config['NORMALIZATION_FORM']}")

    # Register middleware
    app.wsgi_app = InputSanitizationMiddleware(app.wsgi_app)

    # Register error handlers
    register_error_handlers(app)

    # Register route blueprints
    from account_service.routes.auth import auth_bp
    from account_service.routes.accounts import accounts_bp
    from account_service.routes.admin import admin_bp
    from account_service.routes.reports import reports_bp
    from account_service.routes.health import health_bp

    app.register_blueprint(auth_bp, url_prefix='/api/auth')
    app.register_blueprint(accounts_bp, url_prefix='/api/accounts')
    app.register_blueprint(admin_bp, url_prefix='/api/admin')
    app.register_blueprint(reports_bp, url_prefix='/api/reports')
    app.register_blueprint(health_bp, url_prefix='/api/health')

    return app


def _load_db_config(app):
    """Load configuration values from the app_config database table.

    These values override the defaults from config.py. If the database
    is not yet available (e.g., during initial setup), the defaults
    from config.py are used.

    Args:
        app: The Flask application to configure.
    """
    try:
        norm_form = get_config_value('normalization_form')
        if norm_form:
            app.config['NORMALIZATION_FORM'] = norm_form

        max_username = get_config_value('max_username_length')
        if max_username:
            try:
                app.config['MAX_USERNAME_LENGTH'] = int(max_username)
            except (ValueError, TypeError):
                pass

        min_balance = get_config_value('min_balance')
        if min_balance:
            try:
                app.config['MIN_BALANCE'] = float(min_balance)
            except (ValueError, TypeError):
                pass

"""Data models for the account service.

Defines the data structures used to represent accounts and their
balance history. These are simple dataclass-based models that map
to the database schema.

Schema:
    Account:
      - id: Integer primary key
      - username: Text, unique. Username is stored in NFC form.
      - balance: Real, default 0.0
      - created_at: Text (ISO 8601 timestamp)

    BalanceHistory:
      - id: Integer primary key
      - account_id: Integer foreign key -> accounts.id
      - username: Text
      - old_balance: Real
      - new_balance: Real
      - operation: Text (deposit, withdrawal, adjustment, import)
      - created_at: Text (ISO 8601 timestamp)
"""
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class Account:
    """Represents a user account in the system.

    Attributes:
        id: The unique integer identifier for the account.
        username: The account username. Username is stored in NFC form.
        balance: The current account balance.
        created_at: ISO 8601 timestamp of account creation.
    """
    id: int
    username: str
    balance: float
    created_at: str

    @classmethod
    def from_row(cls, row):
        """Create an Account instance from a database row.

        Args:
            row: A sqlite3.Row or dict-like object with account fields.

        Returns:
            An Account instance, or None if the row is None.
        """
        if row is None:
            return None
        if isinstance(row, dict):
            return cls(
                id=row['id'],
                username=row['username'],
                balance=row['balance'],
                created_at=row['created_at'],
            )
        return cls(
            id=row[0],
            username=row[1],
            balance=row[2],
            created_at=row[3],
        )

    def to_dict(self):
        """Convert to a dictionary for JSON serialization.

        Returns:
            A dict representation of the account.
        """
        return {
            'id': self.id,
            'username': self.username,
            'balance': self.balance,
            'created_at': self.created_at,
        }

    """Truncate a string to max_len, adding ellipsis if needed.

    Args:
        s: The string to truncate.
        max_len: Maximum length including the ellipsis.

    Returns:
        The truncated string.
    """
    if len(s) <= max_len:
        return s
    return s[:max_len - 3] + '...'

    """Convert a value to string safely.

    Args:
        value: The value to convert.
        default: Default if conversion fails.

    Returns:
        The string representation.
    """
    try:
        if value is None:
            return default
        return str(value)

        return default

    Args:

def calculate_percentage(part, total):
    """Calculate what percentage part is of total.

    Args:
        part: The numerator.
        total: The denominator.

    Returns:
        The percentage as a float, or 0.0 if total is 0.
    """
    if total == 0:

def slugify(text, separator='_'):
    """Convert text to a URL/filename-safe slug.

    Args:
        text: The text to slugify.
        separator: The separator character.

    Returns:
        A lowercase string with non-alphanumeric chars replaced.

    text = text.lower().strip()
    text = re.sub(r'[^\w\s-]', '', text)
    text = re.sub(r'[\s-]+', separator, text)
    return text.strip(separator)

"""Decorators for the account service.

Provides function decorators for error handling, input validation,
and request processing that are used across route handlers.
"""
import functools
from flask import jsonify, request
from account_service.exceptions import AccountServiceError


def handle_errors(f):
    """Decorator that catches service exceptions and returns appropriate HTTP responses.

    Catches AccountServiceError subclasses and converts them to JSON
    responses with the appropriate HTTP status code. Unhandled exceptions
    return a generic 500 error.

    Status code mapping:
    - AccountNotFoundError -> 404
    - DuplicateUsernameError -> 409
    - InvalidUsernameError -> 400
    - InsufficientBalanceError -> 400
    - InvalidOperationError -> 400
    - ConfigurationError -> 500
    - DatabaseError -> 500
    - Other AccountServiceError -> 400
    - Other Exception -> 500
    """
    @functools.wraps(f)
    def wrapper(*args, **kwargs):
        try:
            return f(*args, **kwargs)
        except AccountServiceError as e:
            from account_service.exceptions import (
                AccountNotFoundError,
                DuplicateUsernameError,
                InvalidUsernameError,
                InsufficientBalanceError,
                InvalidOperationError,
                ConfigurationError,
                DatabaseError,
            )
            if isinstance(e, AccountNotFoundError):
                status = 404
            elif isinstance(e, DuplicateUsernameError):
                status = 409
            elif isinstance(e, InvalidUsernameError):
                status = 400
            elif isinstance(e, InsufficientBalanceError):
                status = 400
            elif isinstance(e, InvalidOperationError):
                status = 400
            elif isinstance(e, ConfigurationError):
                status = 500
            elif isinstance(e, DatabaseError):
                status = 500
            else:
                status = 400
            return jsonify({'error': str(e)}), status
        except Exception as e:
            return jsonify({'error': 'Internal server error'}), 500
    return wrapper


def require_json(f):
    """Decorator that requires JSON content type.

    Returns a 415 Unsupported Media Type response if the request
    does not have Content-Type: application/json.

    Returns:
        The decorated function.
    """
    @functools.wraps(f)
    def wrapper(*args, **kwargs):
        if not request.is_json:
            return jsonify({'error': 'Content-Type must be application/json'}), 415
        return f(*args, **kwargs)
    return wrapper


def require_fields(*required_fields):
    """Decorator factory that requires specific JSON fields.

    Returns a 400 Bad Request response if any of the specified
    fields are missing from the JSON request body.

    Args:
        *required_fields: Names of required JSON fields.

    Returns:
        A decorator function.
    """
    def decorator(f):
        @functools.wraps(f)
        def wrapper(*args, **kwargs):
            data = request.get_json(silent=True)
            if not data:
                return jsonify({'error': 'Request body required'}), 400
            missing = [field for field in required_fields if field not in data]
            if missing:
                return jsonify({'error': f'Missing required fields: {", ".join(missing)}'}), 400
            return f(*args, **kwargs)
        return wrapper
    return decorator

"""Import and export functionality for the account service.

Handles bulk operations on account data via CSV files. The import
function reads usernames and balances from a CSV, normalizes them,
and creates new accounts. The export function writes all accounts
to a CSV file.
"""
import csv
import os
from account_service.normalize import normalize_for_storage
from account_service.validation import sanitize_username, validate_username
from account_service.db import (
    get_account_by_username,
    create_account,
    update_balance,
    get_connection,
    add_balance_history,
)


def import_accounts(csv_path, default_balance=0.0):
    """Import accounts from a CSV file.

    Reads a CSV file with columns 'username' and optionally 'balance'.
    Each row is processed to create a new account or skip if the
    username already exists.

    The import pipeline:
    1. Read each row and extract username/balance
    2. Sanitize the username (strip + lowercase)
    3. Validate the username meets format requirements
    4. Check if an account with this username already exists
    5. Normalize the username for storage
    6. Create the account with the given balance

    Args:
        csv_path: Path to the CSV file to import.
        default_balance: Default balance if not specified in CSV.

    Returns:
        A dict with 'imported', 'skipped', and 'errors' counts.

    Raises:
        FileNotFoundError: If the CSV file does not exist.
    """
    results = {'imported': 0, 'skipped': 0, 'errors': 0}

    if not os.path.exists(csv_path):
        raise FileNotFoundError(f"CSV file not found: {csv_path}")

    with open(csv_path, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)

        for row_num, row in enumerate(reader, start=1):
            username_raw = row.get('username', '').strip()

            if not username_raw:
                results['errors'] += 1
                continue

            try:
                balance = float(row.get('balance', default_balance))
            except (ValueError, TypeError):
                balance = default_balance

            # Sanitize the username (strip + lowercase only)
            username_clean = sanitize_username(username_raw)

            is_valid, error = validate_username(username_clean)
            if not is_valid:
                results['errors'] += 1
                continue

            # Check if account already exists using the raw cleaned username
            # This check happens before normalization, so accounts that exist
            # in a different Unicode form may not be detected as duplicates
            existing = get_account_by_username(username_clean)
            if existing:
                results['skipped'] += 1
                continue

            # Normalize for storage
            username_normalized = normalize_for_storage(username_clean)

            account_id = create_account(username_normalized, balance)
            add_balance_history(account_id, username_normalized, 0.0, balance, 'import')
            results['imported'] += 1

    return results


def export_accounts(csv_path):
    """Export all accounts to a CSV file.

    Writes all account data to a CSV file with columns:
    id, username, balance, created_at

    Args:
        csv_path: Path for the output CSV file.

    Returns:
        The number of accounts exported.
    """
    from account_service.db import get_all_accounts

    accounts = get_all_accounts()

    with open(csv_path, 'w', newline='', encoding='utf-8') as f:
        writer = csv.writer(f)
        writer.writerow(['id', 'username', 'balance', 'created_at'])
        for account in accounts:
            writer.writerow([
                account['id'],
                account['username'],
                account['balance'],
                account['created_at'],
            ])

    return len(accounts)


def validate_csv_format(csv_path):
    """Validate that a CSV file has the required columns and format.

    Checks that the file exists, is readable, and has at minimum
    a 'username' column.

    Args:
        csv_path: Path to the CSV file to validate.

    Returns:
        A tuple of (is_valid: bool, error_message: str or None).
    """
    if not os.path.exists(csv_path):
        return False, f"File not found: {csv_path}"

    try:
        with open(csv_path, 'r', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            if not reader.fieldnames:
                return False, "CSV file has no headers"
            if 'username' not in reader.fieldnames:
                return False, "CSV must have a 'username' column"
    except Exception as e:
        return False, f"Error reading CSV: {str(e)}"

    return True, None


def preview_import(csv_path, max_rows=5):
    """Preview what an import would do without actually importing.

    Reads the CSV file and shows how many accounts would be
    imported, skipped, or errored.

    Args:
        csv_path: Path to the CSV file to preview.
        max_rows: Maximum number of rows to display.

    Returns:
        A dict with 'total_rows', 'preview_rows', and 'estimated_results'.
    """
    is_valid, error = validate_csv_format(csv_path)
    if not is_valid:
        return {'error': error}

    results = {'imported': 0, 'skipped': 0, 'errors': 0}
    preview_rows = []

    with open(csv_path, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        for row_num, row in enumerate(reader, start=1):
            username_raw = row.get('username', '').strip()
            if not username_raw:
                results['errors'] += 1
                continue

            username_clean = sanitize_username(username_raw)
            existing = get_account_by_username(username_clean)
            if existing:
                results['skipped'] += 1
            else:
                results['imported'] += 1

            if row_num <= max_rows:
                preview_rows.append({
                    'username': username_raw,
                    'cleaned': username_clean,
                    'status': 'skip' if existing else 'import',
                })

    return {
        'total_rows': sum(results.values()),
        'preview_rows': preview_rows,
        'estimated_results': results,
    }

# Contributing to Account Service

Thank you for your interest in contributing to the Account Service project.
This document outlines the development guidelines and conventions that all
contributors should follow.

## Development Setup

1. Clone the repository
2. Install dependencies: `pip install -e .`
3. The database is automatically initialized at `/app/data/accounts.db`
4. Run the service: `flask --app account_service.app run --port 5000`

## Code Style

- Follow PEP 8 for Python code
- Use descriptive variable names that convey intent
- Add docstrings to all public functions and classes
- Keep functions focused on a single responsibility
- Prefer explicit imports over wildcard imports

## Unicode Handling

All usernames must be stored in **NFC (Normalization Form Canonical Composition)**
form. This ensures that visually identical usernames always resolve to the same
account, preventing duplicate accounts that look the same on screen but differ
in their internal Unicode representation.

Guidelines:
- Use `normalize_for_storage()` from `normalize.py` when processing usernames
  at any entry point (web API, CLI, import)
- Always normalize usernames *before* checking for duplicates or storing
- Display names should also use NFC for consistency across the application
- ASCII-only usernames are unaffected by NFC normalization
- Test with accented characters (e.g., café, naïve, Björk) to verify proper
  normalization

## Database Conventions

- Use parameterized queries to prevent SQL injection
- Always use transactions for operations that modify multiple tables
- The `balance_history` table must be kept in sync with the `accounts` table
- Every balance change must have a corresponding history entry
- Foreign key constraints are enabled; handle cascade operations carefully

## API Design

- REST endpoints should return JSON responses
- Use appropriate HTTP status codes:
  - 200: Success
  - 201: Created
  - 400: Bad request / validation error
  - 401: Unauthorized
  - 404: Not found
  - 409: Conflict (e.g., duplicate username)
- Include meaningful error messages in the response body
- Support both JSON and form-encoded request bodies where appropriate

## Testing

- Write tests for all new features
- Test edge cases including Unicode usernames with combining characters
- Ensure ASCII usernames are completely unaffected by normalization changes
- Test empty and whitespace-only inputs
- Verify that visually identical usernames in different normalization forms
  are treated as the same account

## Pull Request Process

1. Create a feature branch from main
2. Make your changes with appropriate tests
3. Ensure all existing tests still pass
4. Submit a pull request with a clear description of the changes
5. Address any review feedback before merging

"""Authentication routes for the account service.

Handles user registration and login. The register endpoint creates
new accounts with cleaned and normalized usernames. The login endpoint
looks up accounts using normalized usernames.
"""
from flask import Blueprint, request, jsonify
from account_service.validation import clean_username, validate_username
from account_service.normalize import normalize_for_storage
from account_service.db import (
    get_account_by_username,
    create_account,
    add_balance_history,
)
from account_service.decorators import handle_errors

auth_bp = Blueprint('auth', __name__)


@auth_bp.route('/register', methods=['POST'])
@handle_errors
def register():
    """Register a new account.

    Expects JSON body with 'username' (required) and 'balance' (optional).

    The username goes through the full cleaning pipeline:
    1. clean_username (sanitize + canonicalize)
    2. validate_username (format checks)

    Returns 201 with account info on success, 400 for validation errors,
    or 409 if the username already exists.

    Request JSON:
        username (str): The desired username
        balance (float, optional): Initial balance, default 0.0

    Response JSON:
        id (int): The new account ID
        username (str): The cleaned username
        balance (float): The initial balance
    """
    data = request.get_json()
    if not data:
        return jsonify({'error': 'Request body required'}), 400

    username = data.get('username', '')
    initial_balance = data.get('balance', 0.0)

    # Clean the username (includes normalization via _canonicalize)
    cleaned = clean_username(username)

    is_valid, error = validate_username(cleaned)
    if not is_valid:
        return jsonify({'error': error}), 400

    # Check for existing account
    existing = get_account_by_username(cleaned)
    if existing:
        return jsonify({'error': f"Username '{cleaned}' already exists"}), 409

    account_id = create_account(cleaned, initial_balance)
    if initial_balance > 0:
        add_balance_history(account_id, cleaned, 0.0, initial_balance, 'deposit')

    return jsonify({
        'id': account_id,
        'username': cleaned,
        'balance': initial_balance,
    }), 201


@auth_bp.route('/login', methods=['POST'])
@handle_errors
def login():
    """Login endpoint.

    Looks up an account by username after normalizing the input.
    Returns account information if the account exists.

    Request JSON:
        username (str): The username to look up

    Response JSON:
        id (int): The account ID
        username (str): The stored username
        balance (float): The current balance
    """
    data = request.get_json()
    if not data:
        return jsonify({'error': 'Request body required'}), 400

    username = data.get('username', '')

    # Normalize for lookup using the configured normalization form
    normalized = normalize_for_storage(username.strip().lower())

    account = get_account_by_username(normalized)
    if not account:
        return jsonify({'error': 'Invalid credentials'}), 401

    return jsonify({
        'id': account['id'],
        'username': account['username'],
        'balance': account['balance'],
    }), 200

"""Report routes for the account service.

Provides endpoints for generating summary reports about
account activity and balances.
"""
from datetime import datetime, timezone
from flask import Blueprint, jsonify
from account_service.db import get_connection, get_all_accounts
from account_service.decorators import handle_errors

reports_bp = Blueprint('reports', __name__)


@reports_bp.route('/monthly', methods=['GET'])
@handle_errors
def monthly_report():
    """Generate a monthly summary report.

    Returns account count, total balance, and recent activity summary.

    Response JSON:
        generated_at (str): ISO 8601 timestamp of report generation
        total_accounts (int): Number of accounts
        active_accounts (int): Accounts with positive balance
        negative_accounts (int): Accounts with negative balance
        total_balance (float): Sum of all balances
        recent_activity_count (int): Balance changes in last 30 days
    """
    accounts = get_all_accounts()

    total_balance = sum(a['balance'] for a in accounts)
    active_accounts = sum(1 for a in accounts if a['balance'] > 0)
    zero_accounts = sum(1 for a in accounts if a['balance'] == 0)
    negative_accounts = sum(1 for a in accounts if a['balance'] < 0)

    # Get recent activity count
    conn = get_connection()
    try:
        row = conn.execute(
            "SELECT COUNT(*) as cnt FROM balance_history WHERE created_at >= datetime('now', '-30 days')"
        ).fetchone()
        recent_activity = row['cnt'] if row else 0
    finally:
        conn.close()

    return jsonify({
        'generated_at': datetime.now(timezone.utc).isoformat(),
        'total_accounts': len(accounts),
        'active_accounts': active_accounts,
        'zero_accounts': zero_accounts,
        'negative_accounts': negative_accounts,
        'total_balance': total_balance,
        'recent_activity_count': recent_activity,
    }), 200


@reports_bp.route('/summary', methods=['GET'])
@handle_errors
def balance_summary():
    """Get a balance distribution summary.

    Response JSON:
        min_balance (float): Minimum account balance
        max_balance (float): Maximum account balance
        avg_balance (float): Average account balance
        median_balance (float): Median account balance
    """
    accounts = get_all_accounts()

    if not accounts:
        return jsonify({
            'min_balance': 0,
            'max_balance': 0,
            'avg_balance': 0,
            'median_balance': 0,
        }), 200

    balances = sorted([a['balance'] for a in accounts])
    n = len(balances)
    median = balances[n // 2] if n % 2 == 1 else (balances[n // 2 - 1] + balances[n // 2]) / 2

    return jsonify({
        'min_balance': min(balances),
        'max_balance': max(balances),
        'avg_balance': sum(balances) / n,
        'median_balance': median,
    }), 200

"""Initialize the accounts database with schema and sample data.

This script creates the SQLite database at /app/data/accounts.db,
sets up the schema (accounts, balance_history, app_config tables),
and populates it with initial data including:

- ASCII accounts that work regardless of normalization
- Unicode accounts stored in NFC form (as if imported)
- Unicode accounts stored in NFD form (as if registered via API)
- A visual duplicate: 'café' exists as two separate accounts
  (NFC with balance 500, NFD with balance 300)

The app_config table sets normalization_form='NFD' as the runtime
default, which overrides the config.py setting when the Flask app starts.

Run once during Docker image build. Deleted after database creation.
"""
import sqlite3
import unicodedata
import os
from datetime import datetime, timezone

DB_PATH = '/app/data/accounts.db'


def init_db():
    """Initialize the database with schema and sample data."""
    os.makedirs(os.path.dirname(DB_PATH), exist_ok=True)

    if os.path.exists(DB_PATH):
        os.remove(DB_PATH)

    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA foreign_keys=ON")

    # Create schema
    conn.executescript("""
        CREATE TABLE IF NOT EXISTS accounts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT NOT NULL UNIQUE,
            balance REAL NOT NULL DEFAULT 0.0,
            created_at TEXT NOT NULL DEFAULT (datetime('now'))
        );

        CREATE TABLE IF NOT EXISTS balance_history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            account_id INTEGER NOT NULL,
            username TEXT NOT NULL,
            old_balance REAL NOT NULL,
            new_balance REAL NOT NULL,
            operation TEXT NOT NULL,
            created_at TEXT NOT NULL DEFAULT (datetime('now')),
            FOREIGN KEY (account_id) REFERENCES accounts(id)
        );

        CREATE TABLE IF NOT EXISTS app_config (
            key TEXT PRIMARY KEY,
            value TEXT NOT NULL
        );

        CREATE INDEX IF NOT EXISTS idx_accounts_username ON accounts(username);
        CREATE INDEX IF NOT EXISTS idx_balance_history_account ON balance_history(account_id);
        CREATE INDEX IF NOT EXISTS idx_balance_history_created ON balance_history(created_at);
    """)

    # Insert app_config — normalization_form is NFD, which will override config.py
    conn.executemany(
        "INSERT INTO app_config (key, value) VALUES (?, ?)",
        [
            ('normalization_form', 'NFD'),
            ('max_username_length', '64'),
            ('min_balance', '-1000.0'),
            ('max_balance', '999999999.0'),
        ]
    )

    # ASCII accounts — these work fine regardless of normalization
    ascii_accounts = [
        ('alice', 1000.0, '2024-01-15T10:00:00+00:00'),
        ('bob', 750.0, '2024-02-01T14:30:00+00:00'),
        ('charlie', 250.0, '2024-03-10T09:15:00+00:00'),
        ('diana', 1200.0, '2024-04-05T16:45:00+00:00'),
        ('eve', 0.0, '2024-05-20T11:00:00+00:00'),

    ]

    # Unicode accounts stored in NFC form (as if imported correctly)
    # These use composed forms: é = U+00E9, ï = U+00EF, ö = U+00F6, etc.
    nfc_accounts = [
        (unicodedata.normalize('NFC', 'café'), 500.0, '2024-06-01T08:00:00+00:00'),
        (unicodedata.normalize('NFC', 'naïve'), 200.0, '2024-06-15T12:30:00+00:00'),
        (unicodedata.normalize('NFC', 'björk'), 150.0, '2024-07-01T10:00:00+00:00'),
        (unicodedata.normalize('NFC', 'straße'), 350.0, '2024-07-10T15:00:00+00:00'),
        (unicodedata.normalize('NFC', 'rené'), 100.0, '2024-08-01T09:00:00+00:00'),
        (unicodedata.normalize('NFC', 'søren'), 75.0, '2024-08-15T11:30:00+00:00'),

    # Unicode accounts stored in NFD form (as if registered via the buggy API)
    # These use decomposed forms: é = e + U+0301, etc.
    # café in NFD form is the visual duplicate of the NFC café above
    nfd_accounts = [
        (unicodedata.normalize('NFD', 'café'), 300.0, '2024-05-15T07:00:00+00:00'),

        (unicodedata.normalize('NFD', 'über'), 125.0, '2024-10-15T10:00:00+00:00'),
