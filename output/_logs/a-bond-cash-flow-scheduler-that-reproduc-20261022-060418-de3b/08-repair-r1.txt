#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

#define MAX_DATES 1000
#define MAX_HOLIDAYS 200
#define MAX_LINE 8192

static int is_leap(int y) {
    return (y % 4 == 0 && y % 100 != 0) || (y % 400 == 0);
}

static int days_in_month(int y, int m) {
    static const int dim[] = {31,28,31,30,31,30,31,31,30,31,30,31};
    if (m == 2 && is_leap(y)) return 29;
    return dim[m-1];
}

static int days_from_civil(int y, int m, int d) {
    y -= (m <= 2);
    int era = (y >= 0 ? y : y-399) / 400;
    int yoe = y - era * 400;
    int doy = (153 * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
    int doe = yoe * 365 + yoe/4 - yoe/100 + doy;
    return era * 146097 + doe - 719468;
}

static void civil_from_days(int days, int *y, int *m, int *d) {
    days += 719468;
    int era = (days >= 0 ? days : days - 146096) / 146097;
    int doe = days - era * 146097;
    int yoe = (doe - doe/1460 + doe/36524 - doe/146096) / 365;
    int yr = yoe + era * 400;
    int doy = doe - (365*yoe + yoe/4 - yoe/100);
    int mp = (5*doy + 2) / 153;
    int dd = doy - (153*mp+2)/5 + 1;
    int mm = mp < 10 ? mp + 3 : mp - 9;
    if (mm <= 2) yr++;
    *y = yr;
    *m = mm;
    *d = dd;
}

static int calc_dow(int days) {
    return ((days % 7) + 4 + 7) % 7;
}

static void add_months(int y, int m, int d, int n, int *ry, int *rm, int *rd) {
    int total = y * 12 + (m - 1) + n;
    *ry = total / 12;
    *rm = total % 12;
    if (*rm < 0) { *rm += 12; (*ry)--; }
    *rm += 1;
    int maxd = days_in_month(*ry, *rm);
    *rd = (d > maxd) ? maxd : d;
}

static int parse_date(const char *s, int *y, int *m, int *d) {
    if (strlen(s) != 10 || s[4] != '-' || s[7] != '-') return 0;
    char buf[5];
    strncpy(buf, s, 4); buf[4] = 0; *y = atoi(buf);
    strncpy(buf, s+5, 2); buf[2] = 0; *m = atoi(buf);
    strncpy(buf, s+8, 2); buf[2] = 0; *d = atoi(buf);
    if (*y < 1 || *y > 9999) return 0;
    if (*m < 1 || *m > 12) return 0;
    if (*d < 1 || *d > days_in_month(*y, *m)) return 0;
    return 1;
}

static int hol_count = 0;
static int hol_days[MAX_HOLIDAYS];

static int is_bd(int days) {
    int w = calc_dow(days);
    if (w == 0 || w == 6) return 0;
    for (int i = 0; i < hol_count; i++) {
        if (days == hol_days[i]) return 0;
        int hw = calc_dow(hol_days[i]);
        if (hw == 6 && days == hol_days[i] - 1) return 0;
        if (hw == 0 && days == hol_days[i] + 1) return 0;
    }
    return 1;
}

static int adjust_date(int days, int bdc) {
    if (bdc == 0) return days;
    if (bdc == 1 || bdc == 3) {
        while (!is_bd(days)) days++;
    } else {
        while (!is_bd(days)) days--;
    }
    return days;
}

static double day_count_30u360(int y1, int m1, int d1, int y2, int m2, int d2) {
    if (d1 == 31) d1 = 30;
    if (m1 == 2 && d1 == days_in_month(y1, 2)) d1 = 30;
    if (d2 == 31) d2 = 30;
    int days = 360 * (y2 - y1) + 30 * (m2 - m1) + (d2 - d1);
    return (double)days / 360.0;
}

static double day_count_30e360(int y1, int m1, int d1, int y2, int m2, int d2) {
    if (d1 == 31) d1 = 30;
    if (d2 == 31) d2 = 30;
    if (m1 == 2 && d1 == days_in_month(y1, 2)) d1 = 30;
    int days = 360 * (y2 - y1) + 30 * (m2 - m1) + (d2 - d1);
    return (double)days / 360.0;
}

static double day_count_act365(int y1, int m1, int d1, int y2, int m2, int d2) {
    int start = days_from_civil(y1, m1, d1);
    int end = days_from_civil(y2, m2, d2);
    int actual = end - start;
    double denom = 365.0;
    for (int y = y1; y <= y2; y++) {
        if (is_leap(y)) {
            int feb29 = days_from_civil(y, 2, 29);
            if (feb29 > start && feb29 <= end) {
                denom = 365.25;
                break;
            }
        }
    }
    return (double)actual / denom;
}

static double day_count_actact(int y1, int m1, int d1, int y2, int m2, int d2) {
    int start = days_from_civil(y1, m1, d1);
    int end = days_from_civil(y2, m2, d2);
    int actual = end - start;
    double denom = is_leap(y2) ? 366.0 : 365.0;
    return (double)actual / denom;
}
