def local_to_utc(day, hour, minute, offset_str):
    local_dt = datetime(2024, 3, day, hour, minute, 0)
    sign_char = offset_str[0]
    parts = offset_str[1:].split(':')
    oh = int(parts[0])
    om = int(parts[1]) if len(parts) > 1 else 0
    if sign_char == '-':
        utc_dt = local_dt + timedelta(hours=oh, minutes=om)
    else:
        utc_dt = local_dt - timedelta(hours=oh, minutes=om)
    return utc_dt.strftime('%Y-%m-%dT%H:%M:%SZ')

    1: '-05:00',
    2: '-03:30',
    3: '+00:00',
    4: '+05:30',
    5: '+10:00',

all_shipments = []

    global sid

    tn = f'SHP{sid:06d}'
    offset = CORRECT_OFFSETS[region_id]

    all_shipments.append((sid, tn, weight, status, region_id, utc_str))

"""Timezone helper utilities for shipment processing."""

from datetime import datetime, timedelta


def utc_to_local(utc_str, offset_str):
    """Convert UTC datetime string to local datetime given offset like '-05:00'."""
    dt = datetime.fromisoformat(utc_str.replace('Z', '+00:00'))
    sign = 1 if offset_str[0] == '-' else -1
    parts = offset_str[1:].split(':')
    hours = int(parts[0])
    minutes = int(parts[1]) if len(parts) > 1 else 0
    offset = timedelta(hours=hours, minutes=minutes) * sign
    return dt + offset


def local_date(utc_str, offset_str):
    """Get the local date for a UTC datetime given offset."""
    local = utc_to_local(utc_str, offset_str)
    return local.strftime('%Y-%m-%d')


def format_offset(offset_str):
    """Format offset string for display."""
    return f'UTC{offset_str}'


def parse_offset(offset_str):
    """Parse offset string into total minutes."""
    sign = 1 if offset_str[0] == '-' else -1
    parts = offset_str[1:].split(':')
    hours = int(parts[0])
    minutes = int(parts[1]) if len(parts) > 1 else 0
    return sign * (hours * 60 + minutes)

"""Database utility functions for shipment database."""

import sqlite3


def get_connection(db_path='/app/shipments.db'):
    """Get a database connection with row factory."""
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    return conn


def execute_query(db_path, query, params=None):
    """Execute a query and return all results."""
    conn = get_connection(db_path)
    try:
        cursor = conn.cursor()
        if params:
            cursor.execute(query, params)
        else:
            cursor.execute(query)
        return cursor.fetchall()
    finally:
        conn.close()


def get_table_info(db_path, table_name):
    """Get column info for a table."""
    return execute_query(db_path, f"PRAGMA table_info({table_name})")


def get_view_definitions(db_path):
    """Get all view definitions."""
    return execute_query(db_path, "SELECT name, sql FROM sqlite_master WHERE type='view'")

"""CSV formatting helpers for report generation."""


def format_weight(value, decimal_places=2):
    """Format a weight value for display."""
    if value is None:
        return ''
    return f'{value:.{decimal_places}f}'


def format_count(value):
    """Format a count value for display."""
    return str(int(value))


def format_date(date_str):
    """Format a date string for display."""
    return date_str


def format_report_row(row):
    """Format a complete report row."""
    return [
        format_date(row[0]),
        str(row[1]),
        str(row[2]),
        format_count(row[3]),
        format_weight(row[4]),
        format_weight(row[5]),
    ]

    """Format a value as currency (unused placeholder)."""
    if value is None:
        return ''
    return f'${value:,.2f}'

set -euo pipefail

# Fix 1: Remove the WHERE status != 'cancelled' clause from v_valid_events
cat > /app/queries/03_valid_events.sql << 'EOF'
-- Business rule: only include shipments with valid status for reporting
CREATE VIEW IF NOT EXISTS v_valid_events AS

FROM v_local_dates;

sed -i 's/NULLIF(shipment_count, 0)/NULLIF(measured_count, 0)/' /app/queries/05_report_metrics.sql

# Fix 3: Fix rebuild_views.sh to drop views before recreating them
cat > /app/rebuild_views.sh << 'REBUILD_EOF'
#!/bin/bash
set -euo pipefail

# Drop all existing views in reverse dependency order
for f in $(ls -r /app/queries/*.sql); do
    view_name=$(grep 'CREATE VIEW IF NOT EXISTS' "$f" | head -1 | sed 's/.*CREATE VIEW IF NOT EXISTS //' | sed 's/ .*//')
    if [ -n "$view_name" ]; then
        sqlite3 /app/shipments.db "DROP VIEW IF EXISTS $view_name;" 2>/dev/null || true
    fi
done

# Recreate all views in dependency order
for f in /app/queries/*.sql; do
    sqlite3 /app/shipments.db < "$f"
done

echo "Views rebuilt successfully."
REBUILD_EOF
chmod +x /app/rebuild_views.sh

# Fix 4: Correct the timezone offset for NA-NFL in the database
sqlite3 /app/shipments.db "UPDATE regions SET tz_offset = '-03:30' WHERE code = 'NA-NFL';"

# Rebuild all views with fixed definitions
bash /app/rebuild_views.sh

# Regenerate the report
python3 /app/report.py

    """Run Python code using the image's interpreter and return parsed JSON output."""
    result = subprocess.run(

        for key, exp in expected_map.items():
            r = report_map[key]
            if exp['avg_weight'] is None:
                assert r['avg_weight'] is None, \
                    f"{key}: avg_weight should be empty, got {r['avg_weight']}"
            else:
                assert r['avg_weight'] is not None, \
                    f"{key}: avg_weight should be {exp['avg_weight']}, got empty"
                assert abs(r['avg_weight'] - exp['avg_weight']) < 0.015, \
                    f"{key}: avg_weight={r['avg_weight']}, expected={exp['avg_weight']}"

    def test_total_weight_values(self, expected_map, report_map):
        for key, exp in expected_map.items():
            r = report_map[key]
            if exp['total_weight'] is None:
                assert r['total_weight'] is None, \
                    f"{key}: total_weight should be empty, got {r['total_weight']}"
            else:
                assert r['total_weight'] is not None, \
                    f"{key}: total_weight should be {exp['total_weight']}, got empty"
                assert abs(r['total_weight'] - exp['total_weight']) < 0.015, \
                    f"{key}: total_weight={r['total_weight']}, expected={exp['total_weight']}"

    def test_all_null_group_has_empty_avg(self, expected_report, report_rows):
        all_null_groups = [e for e in expected_report if e['avg_weight'] is None]
        assert len(all_null_groups) > 0, "Test data must contain at least one all-NULL weight group"
        report_map = {(r['local_date'], r['region_code']): r for r in report_rows}
        for exp in all_null_groups:
            key = (exp['local_date'], exp['region_code'])
            assert key in report_map, f"All-NULL group {key} missing from report"
            assert report_map[key]['avg_weight'] is None, \
                f"All-NULL group {key}: avg_weight should be empty, got {report_map[key]['avg_weight']}"

            capture_output=True, text=True
        )
        assert result.stdout.strip() == '-03:30', \
            f"NA-NFL offset must be -03:30, got {result.stdout.strip()}"

    def test_boundary_date_with_correct_offset(self):
        result = subprocess.run(
            [IMAGE_PYTHON, '-c',
             "import sqlite3; conn=sqlite3.connect('/app/shipments.db'); "
             "print(conn.execute(\"SELECT DATE('2024-03-14T02:45:00Z', '-03:30')\").fetchone()[0])"],
            capture_output=True, text=True
        )
        assert result.stdout.strip() == '2024-03-13', \
            "UTC 2024-03-14T02:45:00Z at -03:30 should be March 13"

    def test_wrong_offset_would_give_wrong_date(self):
        """Confirm that the wrong offset (-02:30) would shift the date."""
        result = subprocess.run(
            [IMAGE_PYTHON, '-c',
             "import sqlite3; conn=sqlite3.connect('/app/shipments.db'); "
             "print(conn.execute(\"SELECT DATE('2024-03-14T02:45:00Z', '-02:30')\").fetchone()[0])"],
            capture_output=True, text=True
        )
        assert result.stdout.strip() == '2024-03-14', \
            "With wrong offset -02:30, boundary shipment would appear on March 14"

    def test_nfl_march13_has_shipments(self, report_rows):
        nfl_m13 = [r for r in report_rows
                    if r['region_code'] == 'NA-NFL' and r['local_date'] == '2024-03-13']
        assert len(nfl_m13) > 0, "NA-NFL must have shipments on March 13 in report"
