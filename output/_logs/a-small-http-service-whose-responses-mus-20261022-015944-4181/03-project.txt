"""LinkVault URL shortener - main application."""

import string
import random
from http.server import HTTPServer, BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs

from linkvault.store import (
    init_db, get_link, create_link, get_user,
    update_display_name, get_display_name, create_session
)
from linkvault.auth import set_last_user_cookie, check_auth, verify_password
from linkvault.response import JSONResponse


class LinkVaultHandler(BaseHTTPRequestHandler):

    def do_GET(self):
        path = self.path
        if path.startswith('/go/'):
            self._handle_redirect(path[4:])
        elif path.startswith('/api/shorten'):
            self._handle_shorten()
        elif path == '/dashboard':
            self._handle_dashboard()
        else:
            self._send_error(404, 'Not Found')

    def do_POST(self):
        path = self.path
        if path == '/login':
            self._handle_login()
        elif path == '/api/profile':
            self._handle_profile()
        else:
            self._send_error(404, 'Not Found')

    def _handle_redirect(self, code):
        url = get_link(code)
        if url is None:
            self._send_error(404, 'Not Found')
            return
        self.send_response(302)
        self.send_header('Location', url)
        self.send_header('Content-Length', '0')
        self.end_headers()

    def _handle_shorten(self):
        parsed = urlparse(self.path)
        params = parse_qs(parsed.query)
        url = params.get('url', [None])[0]
        request_id = params.get('request_id', [None])[0]

        if not url:
            self._send_error(400, 'Missing url parameter')
            return

        # Validate URL scheme
        parsed_url = urlparse(url)
        if parsed_url.scheme not in ('http', 'https'):
            self._send_error(400, f'Invalid URL: {url}')
            return

        # Generate a short code
        code = ''.join(random.choices(string.ascii_lowercase + string.digits, k=6))
        user = check_auth(self)
        create_link(code, url, created_by=user)

        headers = {}
        if request_id:
            headers['X-Request-Id'] = request_id

        resp = JSONResponse(
            self, status=201, reason='Created',
            body={'code': code, 'url': url},
            headers=headers
        )
        resp.send()

    def _handle_login(self):
        content_length = int(self.headers.get('Content-Length', 0))
        body = self.rfile.read(content_length).decode('utf-8')
        params = parse_qs(body)
        username = params.get('username', [''])[0]
        password = params.get('password', [''])[0]

        user = get_user(username)
        if user and verify_password(password, user['password_hash']):
            session_id = create_session(username)
            self.send_response(302)
            self.send_header('Set-Cookie', f'session={session_id}; Path=/')
            self.send_header('Location', '/dashboard')
            set_last_user_cookie(self, username)
            self.send_header('Content-Length', '0')
            self.end_headers()
        else:
            self.send_response(401)
            set_last_user_cookie(self, username)
            body_bytes = b'{"error":"invalid credentials"}'
            self.send_header('Content-Type', 'application/json')
            self.send_header('Content-Length', str(len(body_bytes)))
            self.end_headers()
            self.wfile.write(body_bytes)

    def _handle_profile(self):
        user = check_auth(self)
        if user is None:
            self._send_error(401, 'Unauthorized')
            return

        content_length = int(self.headers.get('Content-Length', 0))
        body = self.rfile.read(content_length).decode('utf-8')
        params = parse_qs(body)
        display_name = params.get('display_name', [None])[0]

        if display_name is not None:
            update_display_name(user, display_name)

        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        body_bytes = b'{"ok":true}'
        self.send_header('Content-Length', str(len(body_bytes)))
        self.end_headers()
        self.wfile.write(body_bytes)

    def _handle_dashboard(self):
        user = check_auth(self)
        if user is None:
            self._send_error(401, 'Unauthorized')
            return

        display_name = get_display_name(user) or user
        html = f'<h1>Welcome, {display_name}</h1>'.encode('utf-8')

        self.send_response(200)
        self.send_header('Content-Type', 'text/html; charset=utf-8')
        self.send_header('X-Display-Name', display_name)
        self.send_header('Content-Length', str(len(html)))
        self.end_headers()
        self.wfile.write(html)

    def _send_error(self, code, message):
        self.send_response(code, message)
        self.send_header('Content-Type', 'text/plain')
        body_bytes = f'{code} {message}'.encode('utf-8')
        self.send_header('Content-Length', str(len(body_bytes)))
        self.end_headers()
        self.wfile.write(body_bytes)

    def log_message(self, format, *args):
        pass  # Suppress request logging


def main():
    init_db()
    server = HTTPServer(('0.0.0.0', 8080), LinkVaultHandler)
    print('LinkVault running on port 8080', flush=True)
    server.serve_forever()


if __name__ == '__main__':
    main()

"""Authentication helpers for LinkVault."""

import hashlib


def hash_password(password):
    """Hash a password using SHA-256."""
    return hashlib.sha256(password.encode()).hexdigest()


def verify_password(password, pw_hash):
    """Verify a password against its hash."""
    return hash_password(password) == pw_hash


def set_last_user_cookie(handler, username):
    """Set a last_user cookie so the login form can show the last attempted username."""
    cookie_line = f"Set-Cookie: last_user={username}; Path=/\r\n"
    handler._headers_buffer.append(cookie_line.encode())


def check_auth(handler):
    """Read the session cookie and return the authenticated username, or None."""
    cookie_header = handler.headers.get('Cookie', '')
    if not cookie_header:
        return None
    for part in cookie_header.split(';'):
        part = part.strip()
        if part.startswith('session='):
            session_id = part[len('session='):]
            from linkvault.store import get_session_user
            return get_session_user(session_id)
    return None

"""JSON response builder for LinkVault."""

import json


class JSONResponse:
    """Build and send a complete HTTP JSON response.

    Constructs the full response (status line, headers, body) and writes it
    directly to the handler's wfile.
    """

    def __init__(self, handler, status=200, reason='OK', body=None, headers=None):
        self.handler = handler
        self.status = status
        self.reason = reason
        self.body = body or {}
        self.headers = headers or {}

    def send(self):
        body_bytes = json.dumps(self.body).encode('utf-8')
        self.headers['Content-Type'] = 'application/json'
        self.headers['Content-Length'] = str(len(body_bytes))

        # Build the raw HTTP response
        lines = [f'HTTP/1.1 {self.status} {self.reason}\r\n']
        for name, value in self.headers.items():
            lines.append(f'{name}: {value}\r\n')
        lines.append('\r\n')

        response_bytes = ''.join(lines).encode('latin-1') + body_bytes
        self.handler.wfile.write(response_bytes)
        self.handler.wfile.flush()

"""SQLite storage layer for LinkVault."""

import sqlite3
import os
import hashlib

DB_PATH = '/app/data/linkvault.db'


def get_connection():
    os.makedirs(os.path.dirname(DB_PATH), exist_ok=True)
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = get_connection()
    conn.execute('''CREATE TABLE IF NOT EXISTS users (
        username TEXT PRIMARY KEY,
        password_hash TEXT NOT NULL,
        display_name TEXT NOT NULL
    )''')
    conn.execute('''CREATE TABLE IF NOT EXISTS links (
        code TEXT PRIMARY KEY,
        url TEXT NOT NULL,
        created_by TEXT,
        created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    )''')
    conn.execute('''CREATE TABLE IF NOT EXISTS sessions (
        session_id TEXT PRIMARY KEY,
        username TEXT NOT NULL
    )''')
    # Seed admin user
    pw_hash = hashlib.sha256('admin123'.encode()).hexdigest()
    conn.execute(
        'INSERT OR IGNORE INTO users (username, password_hash, display_name) VALUES (?, ?, ?)',
        ('admin', pw_hash, 'Administrator')
    )
    # Seed sample short links
    conn.execute(
        "INSERT OR IGNORE INTO links (code, url, created_by) VALUES (?, ?, ?)",
        ('abc123', 'https://example.com', 'admin')
    )
    conn.execute(
        "INSERT OR IGNORE INTO links (code, url, created_by) VALUES (?, ?, ?)",
        ('xyz789', 'https://python.org', 'admin')
    )
    conn.commit()
    conn.close()


def get_link(code):
    conn = get_connection()
    row = conn.execute('SELECT url FROM links WHERE code = ?', (code,)).fetchone()
    conn.close()
    return row['url'] if row else None


def create_link(code, url, created_by=None):
    conn = get_connection()
    conn.execute(
        'INSERT INTO links (code, url, created_by) VALUES (?, ?, ?)',
        (code, url, created_by)
    )
    conn.commit()
    conn.close()


def get_user(username):
    conn = get_connection()
    row = conn.execute('SELECT * FROM users WHERE username = ?', (username,)).fetchone()
    conn.close()
    return dict(row) if row else None


def update_display_name(username, display_name):
    conn = get_connection()
    conn.execute(
        'UPDATE users SET display_name = ? WHERE username = ?',
        (display_name, username)
    )
    conn.commit()
    conn.close()


def get_display_name(username):
    conn = get_connection()
    row = conn.execute('SELECT display_name FROM users WHERE username = ?', (username,)).fetchone()
    conn.close()
    return row['display_name'] if row else None


def create_session(username):
    import uuid
    session_id = str(uuid.uuid4())
    conn = get_connection()
    conn.execute(
        'INSERT INTO sessions (session_id, username) VALUES (?, ?)',
        (session_id, username)
    )
    conn.commit()
    conn.close()
    return session_id


def get_session_user(session_id):
    conn = get_connection()
    row = conn.execute('SELECT username FROM sessions WHERE session_id = ?', (session_id,)).fetchone()
    conn.close()
    return row['username'] if row else None
