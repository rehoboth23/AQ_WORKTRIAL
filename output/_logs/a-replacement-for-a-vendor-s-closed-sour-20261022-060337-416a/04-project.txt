#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_ITEMS 1024
#define MAX_LINE 4096

typedef struct {
    char category[64];
    long long quantity;
    long long price_cents;
    long long ext_cents;
    long long qty_disc_cents;
} Item;

static long long rhaz(long long a, long long b, long long d) {
    if (d == 0) return 0;
    long long prod = a * b;
    long long q = prod / d;
    long long r = prod % d;
    if (prod >= 0) {
        if (2 * r >= d) q++;
    } else {
        if (2 * r <= -d) q--;
    }
    return q;
}

static long long parse_price(const char *s) {
    int neg = 0;
    const char *p = s;
    while (*p == ' ' || *p == '\t') p++;
    if (*p == '-') { neg = 1; p++; }
    long long whole = 0;
    while (*p >= '0' && *p <= '9') {
        whole = whole * 10 + (*p - '0');
        p++;
    }
    long long frac = 0;
    int fd = 0;
    if (*p == '.') {
        p++;
        while (*p >= '0' && *p <= '9' && fd < 2) {
            frac = frac * 10 + (*p - '0');
            fd++;
            p++;
        }
    }
    long long cents = whole * 100 + (fd == 1 ? frac * 10 : frac);
    if (neg) cents = -cents;
    return cents;
}

static int tax_rate_bp(const char *s) {
    if (strcmp(s, "CA") == 0) return 725;
    if (strcmp(s, "TX") == 0) return 625;
    if (strcmp(s, "NY") == 0) return 800;
    if (strcmp(s, "FL") == 0) return 600;
    if (strcmp(s, "WA") == 0) return 650;
    if (strcmp(s, "OR") == 0) return 0;
    if (strcmp(s, "MA") == 0) return 625;
    if (strcmp(s, "IL") == 0) return 625;
    if (strcmp(s, "PA") == 0) return 600;
    if (strcmp(s, "OH") == 0) return 575;
    return -1;
}

static int ship_idx(const char *s) {
    if (strcmp(s, "standard") == 0) return 0;
    if (strcmp(s, "express") == 0) return 1;
    if (strcmp(s, "overnight") == 0) return 2;
    return -1;
}

static const long long ship_tbl[3][5] = {
    {500, 850, 1200, 1550, 2000},
    {1000, 1500, 2000, 2500, 3200},
    {2000, 2800, 3800, 4800, 6000}
};

static int pay_bp(const char *s) {
    if (strcmp(s, "credit_card") == 0) return 250;
    if (strcmp(s, "debit") == 0) return 0;
    if (strcmp(s, "ach") == 0) return -50;
    if (strcmp(s, "cash") == 0) return 0;
    if (strcmp(s, "gift_card") == 0) return 100;
    return -1;
}

static int food_ex(const char *s) {
    return strcmp(s, "CA") == 0 || strcmp(s, "NY") == 0 ||
           strcmp(s, "MA") == 0 || strcmp(s, "PA") == 0;
}

static int cloth_ex(const char *s) {
    return strcmp(s, "PA") == 0;
}

static int med_ex(const char *s) {
    return strcmp(s, "WA") != 0;
}

static int ship_taxed(const char *s) {
    return strcmp(s, "TX") == 0 || strcmp(s, "FL") == 0 || strcmp(s, "OH") == 0;
}

static void fmt(long long c, char *buf) {
    int neg = c < 0;
    if (neg) c = -c;
    if (neg)
        sprintf(buf, "-%lld.%02lld", c / 100, c % 100);
    else
        sprintf(buf, "%lld.%02lld", c / 100, c % 100);
}

int main(void) {
    char state[64] = ""; int state_set = 0;
    char type[64] = ""; int type_set = 0;
    int texempt = 0, texempt_set = 0;
    char pay[64] = ""; int pay_set = 0;
    char smeth[64] = ""; int smeth_set = 0;
    int zone = 0, zone_set = 0;
    char coup[64] = ""; int coup_set = 0;
    Item items[MAX_ITEMS];
    int ni = 0;
    char line[MAX_LINE];

    while (fgets(line, sizeof(line), stdin)) {
        int len = strlen(line);
        while (len > 0 && (line[len-1] == '\n' || line[len-1] == '\r'))
            line[--len] = '\0';
        if (len == 0) continue;

        if (strncmp(line, "STATE=", 6) == 0) {
            state_set = 1;
            strncpy(state, line + 6, 63); state[63] = '\0';
        } else if (strncmp(line, "TYPE=", 5) == 0) {
            type_set = 1;
            strncpy(type, line + 5, 63); type[63] = '\0';
        } else if (strncmp(line, "TAX_EXEMPT=", 11) == 0) {
            texempt_set = 1;
            texempt = (strcmp(line + 11, "true") == 0);
        } else if (strncmp(line, "PAYMENT=", 8) == 0) {
            pay_set = 1;
            strncpy(pay, line + 8, 63); pay[63] = '\0';
        } else if (strncmp(line, "SHIP_METHOD=", 12) == 0) {
            smeth_set = 1;
            strncpy(smeth, line + 12, 63); smeth[63] = '\0';
        } else if (strncmp(line, "SHIP_ZONE=", 10) == 0) {
            zone_set = 1;
            zone = atoi(line + 10);
        } else if (strncmp(line, "COUPON=", 7) == 0) {
            coup_set = 1;
            strncpy(coup, line + 7, 63); coup[63] = '\0';
        } else if (strncmp(line, "ITEM ", 5) == 0) {
            if (ni < MAX_ITEMS) {
                char cat[64], ps[64];
                long long q;
                if (sscanf(line + 5, "%63s %lld %63s", cat, &q, ps) == 3) {
                    strncpy(items[ni].category, cat, 63);
                    items[ni].category[63] = '\0';
                    items[ni].quantity = q;
                    items[ni].price_cents = parse_price(ps);
                    items[ni].ext_cents = items[ni].quantity * items[ni].price_cents;
                    items[ni].qty_disc_cents = 0;
                    ni++;
                }
            }
        }
    }

    if (!state_set) { fprintf(stderr, "ERROR: missing STATE\n"); return 1; }
    if (!type_set) { fprintf(stderr, "ERROR: missing TYPE\n"); return 1; }
    if (!pay_set) { fprintf(stderr, "ERROR: missing PAYMENT\n"); return 1; }
    if (!smeth_set) { fprintf(stderr, "ERROR: missing SHIP_METHOD\n"); return 1; }
    if (!zone_set) { fprintf(stderr, "ERROR: missing SHIP_ZONE\n"); return 1; }
    if (ni == 0) { fprintf(stderr, "ERROR: no items\n"); return 1; }
    if (tax_rate_bp(state) < 0) { fprintf(stderr, "ERROR: unknown state\n"); return 1; }
    if (strcmp(type, "retail") != 0 && strcmp(type, "wholesale") != 0 &&
        strcmp(type, "government") != 0) {
        fprintf(stderr, "ERROR: unknown type\n"); return 1;
    }
    if (pay_bp(pay) < 0) { fprintf(stderr, "ERROR: unknown payment method\n"); return 1; }
    if (ship_idx(smeth) < 0) { fprintf(stderr, "ERROR: unknown shipping method\n"); return 1; }
    if (zone < 1 || zone > 5) { fprintf(stderr, "ERROR: invalid zone\n"); return 1; }
    for (int i = 0; i < ni; i++)
        if (items[i].quantity < 0) { fprintf(stderr, "ERROR: negative quantity\n"); return 1; }
    for (int i = 0; i < ni; i++)
        if (items[i].price_cents < 0) { fprintf(stderr, "ERROR: negative price\n"); return 1; }

    long long tqd = 0;
    for (int i = 0; i < ni; i++) {
        int p = 0;
        if (items[i].quantity >= 50) p = 10;
        else if (items[i].quantity >= 10) p = 5;
        items[i].qty_disc_cents = rhaz(items[i].ext_cents, p, 100);
        tqd += items[i].qty_disc_cents;
    }

    long long sub = 0;
    for (int i = 0; i < ni; i++) sub += items[i].ext_cents;

    long long pq = sub - tqd;

    int bp = 0;
    int ws = (strcmp(type, "wholesale") == 0);
    if (ws) {
        if (pq >= 50000) bp = 5;
        else if (pq >= 25000) bp = 3;
    } else {
        if (pq >= 100000) bp = 5;
        else if (pq >= 50000) bp = 3;
    }
    long long bd = rhaz(pq, bp, 100);

    long long pb = pq - bd;

    long long cd = 0;
    int fs = 0;
    if (coup_set && coup[0] != '\0') {
        if (strcmp(coup, "SAVE10") == 0) {
            cd = rhaz(pb, 10, 100);
        } else if (strcmp(coup, "SAVE20") == 0) {
            if (pb >= 10000) cd = rhaz(pb, 20, 100);
        } else if (strcmp(coup, "DOLLAR5") == 0) {
            if (pb >= 5000) cd = 500;
        } else if (strcmp(coup, "DOLLAR25") == 0) {
            if (pb >= 20000) cd = 2500;
        } else if (strcmp(coup, "FREESHIP") == 0) {
            fs = 1;
        }
    }

    long long ds = pb - cd;

    int si = ship_idx(smeth);
    long long shp = ship_tbl[si][zone - 1];
    if (fs) shp = 0;

    long long fuel = rhaz(shp, 10, 100);
    if (fuel > 500) fuel = 500;

    long long tq = 0;
    for (int i = 0; i < ni; i++) tq += items[i].quantity;
    long long hdl = 200 + 50 * tq;
    if (hdl > 1000) hdl = 1000;

    int gov = (strcmp(type, "government") == 0);
    long long sof = 0;
    if (sub < 2500 && !gov) sof = 500;

    long long exm = 0;
    for (int i = 0; i < ni; i++) {
        int ex = 0;
        if (strcmp(items[i].category, "food") == 0 && food_ex(state)) ex = 1;
        if (strcmp(items[i].category, "clothing") == 0 && cloth_ex(state)) ex = 1;
        if (strcmp(items[i].category, "medicine") == 0 && med_ex(state)) ex = 1;
        if (ex) exm += items[i].ext_cents - items[i].qty_disc_cents;
    }

    long long txbl = 0, tax = 0;
    if (gov || texempt) {
        txbl = 0; tax = 0;
    } else {
        txbl = ds - exm;
        if (ship_taxed(state)) txbl += shp;
        tax = rhaz(txbl, tax_rate_bp(state), 10000);
    }

    long long pp = ds + tax + shp + fuel + hdl + sof;
    long long ps = rhaz(pp, pay_bp(pay), 10000);

    long long tot = pp + ps;

    char buf[64];
    fmt(sub, buf); printf("SUBTOTAL=%s\n", buf);
    fmt(tqd, buf); printf("QTY_DISCOUNT=%s\n", buf);
    fmt(bd, buf); printf("BULK_DISCOUNT=%s\n", buf);
    fmt(cd, buf); printf("COUPON_DISCOUNT=%s\n", buf);
    fmt(txbl, buf); printf("TAXABLE=%s\n", buf);
    fmt(tax, buf); printf("TAX=%s\n", buf);
    fmt(shp, buf); printf("SHIPPING=%s\n", buf);
    fmt(fuel, buf); printf("FUEL=%s\n", buf);
    fmt(hdl, buf); printf("HANDLING=%s\n", buf);
    fmt(sof, buf); printf("SMALL_ORDER_FEE=%s\n", buf);
    fmt(ps, buf); printf("PAYMENT_SURCHARGE=%s\n", buf);
    fmt(tot, buf); printf("TOTAL=%s\n", buf);

    return 0;
}
