#!/usr/bin/env python3
"""
Event-sourced inventory ledger for warehouse stock management.

Records RECEIVE, SHIP, RETURN, and CORRECT events in an append-only JSONL
log and maintains a materialised stock snapshot.  The system supports:

  - Late corrections: adjusting a past event's effective quantity after
    the fact (e.g. a shipment count that was misreported).
  - Idempotent processing: if the same event is delivered more than once
    (message redelivery), it is applied exactly once and has no further
    effect on stock.

Data directory layout:
  events.jsonl   Append-only event log, one JSON object per line.
  snapshot.json  Current stock levels {sku: quantity}.
  applied.json   Set of event IDs already applied {event_id: true}.

Usage:
  python3 ledger.py <command> [args...]

Commands:
  receive <sku> <qty> [event_id]    Record received items
  ship <sku> <qty> [event_id]       Record shipped items
  return <sku> <qty> [event_id]     Record returned items
  correct <ref_id> <new_qty> [eid]  Correct a past event's effective qty
  rebuild                           Deduplicate log and rebuild snapshot
  show                              Display current stock levels
  verify                            Check snapshot against full replay
  history                           Show event log (human-readable)
  log                               Dump raw event log (JSON lines)
"""

import json
import os
import sys
import uuid
from datetime import datetime, timezone
from pathlib import Path


# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------

DATA_DIR = Path(os.environ.get("LEDGER_DATA_DIR", "/app/data"))
EVENTS_FILE = DATA_DIR / "events.jsonl"
SNAPSHOT_FILE = DATA_DIR / "snapshot.json"
APPLIED_FILE = DATA_DIR / "applied.json"


# ---------------------------------------------------------------------------
# Utility helpers
# ---------------------------------------------------------------------------

def _now_iso():
    """Current UTC time as an ISO 8601 string."""
    return datetime.now(timezone.utc).isoformat()


def _gen_id():
    """Short unique hex identifier for a new event."""
    return uuid.uuid4().hex[:12]


def _load_json(path, default):
    """Load a JSON file; return *default* if the file does not exist."""
    if path.exists():
        with open(path, "r") as f:
            return json.load(f)
    return default


def _save_json(path, data):
    """Write *data* to a JSON file with sorted keys and indentation."""
    with open(path, "w") as f:
        json.dump(data, f, indent=2, sort_keys=True)


def _append_line(path, obj):
    """Append a JSON object as a single line to *path*."""
    with open(path, "a") as f:
        f.write(json.dumps(obj, sort_keys=True) + "\n")


def _validate_sku(sku):
    """Ensure SKU is a non-empty string."""
    if not isinstance(sku, str) or not sku.strip():
        raise ValueError("SKU must be a non-empty string")
    return sku.strip()


def _validate_positive_int(value, name="value"):
    """Ensure *value* is a positive integer."""
    if not isinstance(value, int) or value <= 0:
        raise ValueError(f"{name} must be a positive integer, got {value!r}")
    return value


def _validate_nonneg_int(value, name="value"):
    """Ensure *value* is a non-negative integer (allowed for corrections)."""
    if not isinstance(value, int) or value < 0:
        raise ValueError(f"{name} must be a non-negative integer, got {value!r}")
    return value


# ---------------------------------------------------------------------------
# Ledger
# ---------------------------------------------------------------------------

class Ledger:
    """Event-sourced inventory ledger.

    Maintains an append-only event log and a materialised stock snapshot.
    Every mutation flows through :meth:`process_event`, which persists the
    event and then applies it to the snapshot if it has not been applied
    before.
    """

    def __init__(self, data_dir=None):
        self.data_dir = Path(data_dir) if data_dir else DATA_DIR
        self.events_file = self.data_dir / "events.jsonl"
        self.snapshot_file = self.data_dir / "snapshot.json"
        self.applied_file = self.data_dir / "applied.json"

        self._ensure_data_dir()
        self.snapshot = _load_json(self.snapshot_file, {})
        self.applied = _load_json(self.applied_file, {})

    # -- directory bootstrap ------------------------------------------------

    def _ensure_data_dir(self):
        """Create the data directory and event log if they don't exist."""
        self.data_dir.mkdir(parents=True, exist_ok=True)
        if not self.events_file.exists():
            self.events_file.touch()

    # -- low-level persistence ----------------------------------------------

    def _persist(self, event):
        """Write *event* to the append-only log for durability.

        Writing before processing guarantees that the event survives a
        crash; a subsequent ``rebuild`` can recover the correct state
        from the log.
        """
        _append_line(self.events_file, event)

    def _is_applied(self, event_id):
        """Return True if *event_id* has already been applied."""
        return event_id in self.applied

    def _mark_applied(self, event_id):
        """Record *event_id* as applied and flush the applied set to disk."""
        self.applied[event_id] = True
        _save_json(self.applied_file, self.applied)

    def _update_snapshot(self, sku, delta):
        """Adjust *sku* stock by *delta* (may be negative)."""
        self.snapshot[sku] = self.snapshot.get(sku, 0) + delta
        _save_json(self.snapshot_file, self.snapshot)

    def _replace_snapshot(self, snapshot):
        """Overwrite the entire snapshot (used by ``rebuild``)."""
        self.snapshot = dict(snapshot)
        _save_json(self.snapshot_file, self.snapshot)

    def _read_log(self):
        """Return every event in the log, in order."""
        events = []
        if not self.events_file.exists():
            return events
        with open(self.events_file, "r") as f:
            for line in f:
                stripped = line.strip()
                if stripped:
                    events.append(json.loads(stripped))
        return events

    # -- event lookup -------------------------------------------------------

    def _find_event(self, event_id):
        """Return the first event whose ``event_id`` matches, or None."""
        for ev in self._read_log():
            if ev["event_id"] == event_id:
                return ev
        return None

    def _get_ref_qty(self, ref_id):
        """Return the recorded quantity of event *ref_id*.

        Looks up the ``qty`` field stored in the original event record.
        For RECEIVE and RETURN events this is the quantity received or
        returned; for SHIP events this is the quantity shipped.
        """
        for ev in self._read_log():
            if ev["event_id"] == ref_id:
                return ev["qty"]
        raise ValueError(f"Referenced event {ref_id} not found")

    # -- event processing ---------------------------------------------------

    def process_event(self, event):
        """Process *event*: persist for durability, then apply if new.

        The persist-then-apply ordering ensures that events are never
        lost, even if the process crashes between persisting and
        applying.  A subsequent ``rebuild`` can always recover.
        """
        self._persist(event)
        if self._is_applied(event["event_id"]):
            return                       # already applied — idempotent
        self._apply(event)
        self._mark_applied(event["event_id"])

    def _apply(self, event):
        """Apply a single event to the snapshot."""
        etype = event["type"]
        sku = event["sku"]
        qty = event.get("qty", 0)

        if etype == "RECEIVE":
            self._update_snapshot(sku, qty)
        elif etype == "SHIP":
            self._update_snapshot(sku, -qty)
        elif etype == "RETURN":
            self._update_snapshot(sku, qty)
        elif etype == "CORRECT":
            self.apply_correction(event)
        else:
            raise ValueError(f"Unknown event type: {etype}")

    def apply_correction(self, event):
        """Apply a CORRECT event to the snapshot.

        A correction changes the effective quantity of the referenced
        event.  The stock delta is the difference between the new and
        old effective quantities, with sign flipped for SHIP events
        (because more shipped means less stock).
        """
        ref_id = event["ref_id"]
        new_qty = event["new_qty"]
        sku = event["sku"]

        ref_event = self._find_event(ref_id)
        if ref_event is None:
            raise ValueError(f"Referenced event {ref_id} not found")
        if ref_event["type"] == "CORRECT":
            raise ValueError("Cannot correct a correction event")

        old_effective = self._get_ref_qty(ref_id)
        delta = new_qty - old_effective

        if ref_event["type"] == "SHIP":
            # More shipped means less stock, so stock delta is inverted
            self._update_snapshot(sku, -delta)
        else:
            # RECEIVE or RETURN: more quantity means more stock
            self._update_snapshot(sku, delta)

    # -- public commands ----------------------------------------------------

    def receive(self, sku, qty, event_id=None):
        """Record a RECEIVE event: items arrived at the warehouse."""
        sku = _validate_sku(sku)
        qty = _validate_positive_int(qty, "qty")
        event = {
            "event_id": event_id or _gen_id(),
            "type": "RECEIVE",
            "sku": sku,
            "qty": qty,
            "timestamp": _now_iso(),
        }
        self.process_event(event)
        return event["event_id"]

    def ship(self, sku, qty, event_id=None):
        """Record a SHIP event: items left the warehouse."""
        sku = _validate_sku(sku)
        qty = _validate_positive_int(qty, "qty")
        event = {
            "event_id": event_id or _gen_id(),
            "type": "SHIP",
            "sku": sku,
            "qty": qty,
            "timestamp": _now_iso(),
        }
        self.process_event(event)
        return event["event_id"]

    def return_item(self, sku, qty, event_id=None):
        """Record a RETURN event: items returned to the warehouse."""
        sku = _validate_sku(sku)
        qty = _validate_positive_int(qty, "qty")
        event = {
            "event_id": event_id or _gen_id(),
            "type": "RETURN",
            "sku": sku,
            "qty": qty,
            "timestamp": _now_iso(),
        }
        self.process_event(event)
        return event["event_id"]

    def correct(self, ref_id, new_qty, event_id=None):
        """Issue a late correction to change the effective quantity of a
        past event.

        *ref_id* identifies the original event; *new_qty* is its new
        effective quantity.  The referenced event must not itself be a
        correction.
        """
        new_qty = _validate_nonneg_int(new_qty, "new_qty")
        ref_event = self._find_event(ref_id)
        if ref_event is None:
            raise ValueError(f"Referenced event {ref_id} not found")
        if ref_event["type"] == "CORRECT":
            raise ValueError("Cannot correct a correction event")
        event = {
            "event_id": event_id or _gen_id(),
            "type": "CORRECT",
            "ref_id": ref_id,
            "sku": ref_event["sku"],
            "new_qty": new_qty,
            "timestamp": _now_iso(),
        }
        self.process_event(event)
        return event["event_id"]

    # -- query helpers ------------------------------------------------------

    def get_stock(self, sku):
        """Return the current stock level for *sku*, or 0 if unknown."""
        return self.snapshot.get(sku, 0)

    def list_skus(self):
        """Return sorted list of SKUs that have stock recorded."""
        return sorted(self.snapshot.keys())

    def count_events(self):
        """Return the number of entries in the event log."""
        return len(self._read_log())

    # -- reconciliation -----------------------------------------------------

    def replay(self):
        """Replay the full event log from scratch.

        Builds a map of effective quantities from all CORRECT events in
        the log (handling chained corrections correctly), then replays
        every non-CORRECT event with its effective quantity.

        Returns the resulting snapshot dict.
        """
        events = self._read_log()

        # Collect corrections per referenced event, in log order
        corrections_by_ref = {}
        for ev in events:
            if ev["type"] == "CORRECT":
                rid = ev["ref_id"]
                corrections_by_ref.setdefault(rid, []).append(ev)

        # Compute the latest effective quantity for each referenced event
        effective = {}
        for rid, corrs in corrections_by_ref.items():
            orig = None
            for ev in events:
                if ev["event_id"] == rid and ev["type"] != "CORRECT":
                    orig = ev
                    break
            if orig is None:
                continue
            eff = orig["qty"]
            for c in corrs:
                eff = c["new_qty"]
            effective[rid] = eff

        # Replay non-CORRECT events using effective quantities
        snapshot = {}
        for ev in events:
            if ev["type"] == "CORRECT":
                continue
            sku = ev["sku"]
            eid = ev["event_id"]
            qty = effective.get(eid, ev["qty"])

            if ev["type"] == "RECEIVE":
                snapshot[sku] = snapshot.get(sku, 0) + qty
            elif ev["type"] == "SHIP":
                snapshot[sku] = snapshot.get(sku, 0) - qty
            elif ev["type"] == "RETURN":
                snapshot[sku] = snapshot.get(sku, 0) + qty

        return snapshot

    def verify(self):
        """Compare the live snapshot against a full replay of the log.

        Returns a dict of discrepancies keyed by SKU (empty dict if the
        snapshot is consistent with the log).
        """
        replayed = self.replay()
        discrepancies = {}
        all_skus = set(self.snapshot) | set(replayed)
        for sku in sorted(all_skus):
            live_val = self.snapshot.get(sku, 0)
            replay_val = replayed.get(sku, 0)
            if live_val != replay_val:
                discrepancies[sku] = {"live": live_val, "replay": replay_val}
        return discrepancies

    def rebuild(self):
        """Deduplicate the event log, then rebuild the snapshot and applied
        set from a clean replay.

        Removes entries whose ``event_id`` appeared earlier in the log,
        leaving exactly one entry per distinct event.
        """
        events = self._read_log()

        # Keep only the first occurrence of each event_id
        seen = set()
        unique = []
        for ev in events:
            eid = ev["event_id"]
            if eid not in seen:
                seen.add(eid)
                unique.append(ev)

        # Rewrite the deduplicated log
        with open(self.events_file, "w") as f:
            for ev in unique:
                f.write(json.dumps(ev, sort_keys=True) + "\n")

        # Rebuild snapshot from the clean log
        replayed = self.replay()
        self._replace_snapshot(replayed)

        # Rebuild applied set
        self.applied = {ev["event_id"]: True for ev in self._read_log()}
        _save_json(self.applied_file, self.applied)

    # -- display commands ---------------------------------------------------

    def show(self):
        """Print the current stock snapshot."""
        if not self.snapshot:
            print("No stock data.")
            return
        print("Current stock levels:")
        for sku in sorted(self.snapshot):
            print(f"  {sku}: {self.snapshot[sku]}")

    def history(self):
        """Print the event log in a human-readable format."""
        events = self._read_log()
        if not events:
            print("No events recorded.")
            return
        print("Event history:")
        for idx, ev in enumerate(events, 1):
            if ev["type"] == "CORRECT":
                print(
                    f"  {idx}. [{ev['event_id']}] "
                    f"CORRECT {ev['ref_id']} -> qty={ev['new_qty']} "
                    f"({ev['sku']})"
                )
            else:
                print(
                    f"  {idx}. [{ev['event_id']}] "
                    f"{ev['type']} {ev['sku']} qty={ev['qty']}"
                )

    def log(self):
        """Print the raw event log as JSON lines."""
        events = self._read_log()
        if not events:
            print("Event log is empty.")
            return
        for ev in events:
            print(json.dumps(ev, sort_keys=True))


# ---------------------------------------------------------------------------
# CLI
# ---------------------------------------------------------------------------

def main():
    if len(sys.argv) < 2:
        print("Event-sourced inventory ledger\n")
        print("Usage: ledger.py <command> [args...]\n")
        print("Commands:")
        print("  receive <sku> <qty> [event_id]    Record items received")
        print("  ship    <sku> <qty> [event_id]    Record items shipped")
        print("  return  <sku> <qty> [event_id]    Record items returned")
        print("  correct <ref_id> <new_qty> [eid]  Correct a past event")
        print("  rebuild                           Deduplicate log & rebuild")
        print("  show                              Display stock levels")
        print("  verify                            Check snapshot vs replay")
        print("  history                           Show event log")
        print("  log                               Dump raw JSON log")
        sys.exit(1)

    cmd = sys.argv[1]
    ledger = Ledger()

    try:
        if cmd == "receive":
            if len(sys.argv) < 4:
                print("Usage: ledger.py receive <sku> <qty> [event_id]",
                      file=sys.stderr)
                sys.exit(1)
            sku = sys.argv[2]
            qty = int(sys.argv[3])
            eid_override = sys.argv[4] if len(sys.argv) > 4 else None
            eid = ledger.receive(sku, qty, eid_override)
            print(f"Recorded RECEIVE {sku} {qty} (event_id={eid})")

        elif cmd == "ship":
            if len(sys.argv) < 4:
                print("Usage: ledger.py ship <sku> <qty> [event_id]",
                      file=sys.stderr)
                sys.exit(1)
            sku = sys.argv[2]
            qty = int(sys.argv[3])
            eid_override = sys.argv[4] if len(sys.argv) > 4 else None
            eid = ledger.ship(sku, qty, eid_override)
            print(f"Recorded SHIP {sku} {qty} (event_id={eid})")

        elif cmd == "return":
            if len(sys.argv) < 4:
                print("Usage: ledger.py return <sku> <qty> [event_id]",
                      file=sys.stderr)
                sys.exit(1)
            sku = sys.argv[2]
            qty = int(sys.argv[3])
            eid_override = sys.argv[4] if len(sys.argv) > 4 else None
            eid = ledger.return_item(sku, qty, eid_override)
            print(f"Recorded RETURN {sku} {qty} (event_id={eid})")

        elif cmd == "correct":
            if len(sys.argv) < 4:
                print("Usage: ledger.py correct <ref_id> <new_qty> [event_id]",
                      file=sys.stderr)
                sys.exit(1)
            ref_id = sys.argv[2]
            new_qty = int(sys.argv[3])
            eid_override = sys.argv[4] if len(sys.argv) > 4 else None
            eid = ledger.correct(ref_id, new_qty, eid_override)
            print(f"Recorded CORRECT {ref_id} -> {new_qty} (event_id={eid})")

        elif cmd == "rebuild":
            ledger.rebuild()
            print("Rebuild complete. Snapshot and log have been reconciled.")

        elif cmd == "show":
            ledger.show()

        elif cmd == "verify":
            disc = ledger.verify()
            if not disc:
                print("OK: snapshot matches replay.")
            else:
                print("DISCREPANCY: snapshot does not match replay!")
                for sku, vals in disc.items():
                    print(f"  {sku}: live={vals['live']} replay={vals['replay']}")

        elif cmd == "history":
            ledger.history()

        elif cmd == "log":
            ledger.log()

        else:
            print(f"Unknown command: {cmd}", file=sys.stderr)
            sys.exit(1)

    except ValueError as exc:
        print(f"Error: {exc}", file=sys.stderr)
        sys.exit(1)
    except Exception as exc:
        print(f"Unexpected error: {exc}", file=sys.stderr)
        sys.exit(1)


if __name__ == "__main__":
    main()

# Inventory Ledger

Event-sourced inventory tracking system for warehouse stock management.

## Overview

The ledger records `RECEIVE`, `SHIP`, `RETURN`, and `CORRECT` events in an
append-only JSONL log and maintains a materialised stock snapshot. It supports:

- **Late corrections** — Adjust the effective quantity of a past event after
  the fact (e.g., a shipment count that was misreported).
- **Idempotent processing** — If the same event is delivered more than once due
  to message redelivery, it is applied exactly once.

## Data Files

All data lives in `/app/data/`:

| File            | Description                                   |
|-----------------|-----------------------------------------------|
| `events.jsonl`  | Append-only event log (one JSON object/line)  |
| `snapshot.json` | Current stock levels `{sku: quantity}`        |
| `applied.json`  | Set of applied event IDs `{event_id: true}`   |

## Commands

```
python3 /app/ledger.py <command> [args...]
```

| Command                            | Description                              |
|------------------------------------|------------------------------------------|
| `receive <sku> <qty> [event_id]`   | Record items received                    |
| `ship <sku> <qty> [event_id]`      | Record items shipped                     |
| `return <sku> <qty> [event_id]`    | Record items returned                    |
| `correct <ref_id> <new_qty> [eid]` | Correct a past event's effective qty     |
| `rebuild`                          | Deduplicate log & rebuild snapshot       |
| `show`                             | Display current stock levels             |
| `verify`                           | Check snapshot against full event replay |
| `history`                          | Show event log (human-readable)          |
| `log`                              | Dump raw event log (JSON lines)          |

## Event Types

- **RECEIVE** `sku qty` — Items arrived at the warehouse.
- **SHIP** `sku qty` — Items left the warehouse.
- **RETURN** `sku qty` — Items returned to the warehouse.
- **CORRECT** `ref_id new_qty` — A past event's effective quantity was
  different from what was originally recorded. The correction adjusts stock
  accordingly.

## Idempotency

Events may be redelivered by the messaging system. The ledger tracks which
event IDs have been applied in `applied.json`. Processing an event whose ID
is already in the applied set has no effect on stock.

## Corrections

A `CORRECT` event changes the effective quantity of its referenced event.
For example, if a `RECEIVE` was recorded as 100 units but only 80 actually
arrived, a correction with `new_qty=80` reduces stock by 20.

The stock adjustment from a correction is the difference between the new
effective quantity and the old effective quantity, with the sign determined
by the referenced event's type (SHIP events have inverted sign because more
shipped means less stock).

## Reconciliation

- **verify** — Replays the entire event log from scratch and compares the
  result against the live snapshot. Reports any discrepancies.
- **rebuild** — Removes duplicate entries from the event log and rebuilds
  both the snapshot and the applied set from a clean replay. Use this when
  the log has been corrupted (e.g., by duplicate entries from redeliveries).

## Consistency

After any sequence of operations, the snapshot must match a full replay of
the event log, and the log must contain exactly one entry for each distinct
event.
